#include <stdio.h>
#include <conio.h>

int global_n;

void checkFactorial() {
    int i;
    long f = 1;
    for(i = 1; i <= global_n; i++) f = f * i;
    printf("Factorial: %ld\n", f);
}

void checkReverse() {
    int temp, rev = 0;
    temp = global_n;
    while(temp > 0) {
        rev = rev * 10 + (temp % 10);
        temp = temp / 10;
    }
    printf("Reverse: %d\n", rev);
}

void checkPalindrome() {
    int temp, rev = 0;
    temp = global_n;
    while(temp > 0) {
        rev = rev * 10 + (temp % 10);
        temp = temp / 10;
    }
    if(rev == global_n) printf("Palindrome: Yes\n");
    else printf("Palindrome: No\n");
}

void checkPrime() {
    int i, isPrime = 1;
    if(global_n < 2) isPrime = 0;
    for(i = 2; i * i <= global_n; i++) {
        if(global_n % i == 0) { isPrime = 0; break; }
    }
    if(isPrime) printf("Prime: Yes\n");
    else printf("Prime: No\n");
}

void checkArmstrong() {
    int temp, rem, sum = 0;
    temp = global_n;
    while(temp > 0) {
        rem = temp % 10;
        sum = sum + (rem * rem * rem);
        temp = temp / 10;
    }
    if(sum == global_n) printf("Armstrong: Yes\n");
    else printf("Armstrong: No\n");
}

void checkFibonacci() {
    int a = 0, b = 1, c, i;
    printf("Fibonacci up to %d terms: ", global_n);
    for(i = 1; i <= global_n; i++) {
        printf("%d ", a);
        c = a + b;
        a = b;
        b = c;
    }
    printf("\n");
}

void main() {
    clrscr();
    printf("Enter a number: ");
    scanf("%d", &global_n);
    checkFactorial();
    checkReverse();
    checkPalindrome();
    checkPrime();
    checkArmstrong();
    checkFibonacci();
    getch();
}
